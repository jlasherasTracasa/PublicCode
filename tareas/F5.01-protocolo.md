## Objetivo

1. Tipos DTO propios (no reexportes los de otros crates): serde con `rename_all = "camelCase"` para
   los campos y enums etiquetados con `#[serde(tag = "tipo")]`.
2. Cada tipo deriva `ts_rs::TS` con `#[ts(export)]`. En `.cargo/config.toml`, `[env]` con
   `TS_RS_EXPORT_DIR = { value = "apps/ui/src/protocol", relative = true }`: al pasar `cargo test`
   se generan los `.ts`.
3. Constantes de nombres de método en `pub mod methods` y `PROTOCOL_VERSION = "0.1.0"`.
4. Un test por tipo que haga ida y vuelta JSON con un ejemplo realista.

## Firmas (cópialas tal cual)

```rust
pub const PROTOCOL_VERSION: &str = "0.1.0";

pub enum SessionState { Running, WaitingApproval, Done, Failed, Stuck, Cancelled }
pub struct SessionInfo { pub id: i64, pub title: String, pub backend: String, pub model: Option<String>,
                         pub sandbox: Option<String>, pub state: SessionState, pub parent: Option<i64>, pub created_ms: i64 }
pub enum TranscriptEntry {                       // #[serde(tag = "tipo")]
    User { text: String }, Assistant { text: String },
    ToolCall { call_id: String, name: String, summary: String },
    ToolResult { call_id: String, output: String, is_error: bool },
    System { text: String },
}
pub struct ChangeDto { pub path: String, pub kind: String, pub additions: u32, pub deletions: u32 }
pub struct ApprovalDto { pub id: u64, pub session: i64, pub action: String, pub reason: String, pub provenance: String }
pub struct TaskDto { pub id: String, pub title: String, pub state: String, pub attempts: u32, pub backend: Option<String>, pub mode: String }
pub struct BackendDto { pub id: String, pub kind: String, pub installed: bool, pub version: Option<String>, pub logged_in: Option<bool> }
pub enum Event {                                 // #[serde(tag = "tipo")]
    SessionUpdated { session: SessionInfo },
    Transcript { session: i64, entry: TranscriptEntry },
    ApprovalRequested { approval: ApprovalDto },
    ApprovalResolved { id: u64, allowed: bool },
    ChangesUpdated { session: i64, changes: Vec<ChangeDto> },
    TaskUpdated { task: TaskDto },
    Usage { session: i64, input_tokens: u64, output_tokens: u64 },
}
pub struct SessionCreateParams { pub task: String, pub backend: Option<String>, pub model: Option<String>,
                                 pub sandbox: Option<String>, pub accept: Option<String>, pub parent: Option<i64> }
pub struct SessionSendParams { pub session: i64, pub text: String }
pub struct RpcRequest { pub jsonrpc: String, pub id: u64, pub method: String, pub params: serde_json::Value }
pub struct RpcError { pub code: i32, pub message: String }
pub struct RpcResponse { pub jsonrpc: String, pub id: u64, pub result: Option<serde_json::Value>, pub error: Option<RpcError> }
pub mod methods {
    pub const SESSION_CREATE: &str = "session.create"; pub const SESSION_SEND: &str = "session.send";
    pub const SESSION_CANCEL: &str = "session.cancel"; pub const SESSION_LIST: &str = "session.list";
    pub const SESSION_TRANSCRIPT: &str = "session.transcript"; pub const CHANGES_LIST: &str = "changes.list";
    pub const CHANGES_DIFF: &str = "changes.diff"; pub const CHANGES_APPLY: &str = "changes.apply";
    pub const CHANGES_DISCARD: &str = "changes.discard"; pub const APPROVALS_LIST: &str = "approvals.list";
    pub const APPROVALS_DECIDE: &str = "approvals.decide"; pub const BACKENDS_DETECT: &str = "backends.detect";
    pub const FLOW_STATUS: &str = "flow.status"; pub const FLOW_STOP: &str = "flow.stop";
    pub const SKILLS_LIST: &str = "skills.list"; pub const SERVER_INFO: &str = "server.info";
}
```
