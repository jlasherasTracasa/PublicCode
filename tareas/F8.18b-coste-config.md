4. Tests en el módulo de tests de `lib.rs`.

## Firmas (cópialas tal cual)

```rust
// crates/pcode-config/src/lib.rs
/// Precio de un modelo en dólares por millón de tokens.
#[derive(Debug, Clone, Copy, Default, PartialEq, serde::Deserialize)]
#[serde(default)]
pub struct PriceCfg {
    #[serde(rename = "entrada")]
    pub input: f64,
    #[serde(rename = "salida")]
    pub output: f64,
    #[serde(rename = "cache")]
    pub cache_read: Option<f64>,
}

// en `pub struct Config`
    /// Precios por modelo (`[precios.<modelo>]`).
    #[serde(rename = "precios")]
    pub prices: BTreeMap<String, PriceCfg>,

// en `pub struct PrefsCfg`
    /// Presupuesto de una sesión en dólares: avisa al 80 % y para al 100 %.
    #[serde(rename = "presupuesto")]
    pub budget_usd: Option<f64>,

impl Config {
    /// Precio del modelo (exacto o por el prefijo más largo).
    pub fn price_for(&self, model: &str) -> Option<PriceCfg>;
}
```
